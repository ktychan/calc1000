\section{Fundamentals and Reviews}
\begin{frame}
  \frametitle{Inverse functions}

  Given a function \(f\) whose domain is \(D\) and range is \(R\), its \hlmain{inverse function} (if it exists) is \hlsupp{the} function \(f^{-1}(x)\) such that
  \[
    f^{-1}(f(x)) = x \text{ on } D \quad\text{and}\quad f(f^{-1}(x)) = x \text{ on } R.
  \]

  \pause
  The two ``\(=\)'' are equalities between functions. The rules and domains on both sides of the equations must match, respectively, for the ``\(=\)'' to be valid.
\end{frame}

\begin{frame}
  \frametitle{The equivalent equation for \(f^{-1}(x)\)}

  If you know \(f(x)\) and need to work with \(y = f^{-1}(x)\), then \hlmain{the equivalent equation} \[x = f(y)\] is often useful.
\end{frame}

\begin{frame}
  \frametitle{Exponentials and logarithms}

  \large

  \[
    \begin{array}{rclcrcl}
      e^{a}e^{b} &=& e^{a + b} \\[1ex]
      (e^{a})^{b} &=& e^{ab} \\[1ex]
      \ln(ab) &=& \ln(a) + \ln(b) 
              &&
      \ln(a {\color{red}/} b) &=& \ln(a) {\,\color{red}-\,} \ln(b) \\[3ex]\pause
      b &=& e^{\ln(b)}
        &&
      b^{x} &=& e^{\ln(b)x}
    \end{array}
  \]
\end{frame}

\begin{frame}
  \frametitle{\(e^{x}\) and \(\ln(x)\) are inverses of each other and ...}
  Exponent rules: \(e^{a} e^{b} = e^{a+b}\) and \((e^{a})^{b} = e^{ab}\).

  \bigskip
  Change base to \(e\).
  \[
    {\color{main}b}^{(\cdots)} = {\color{main}e}^{{\color{main}\ln(b)}(\cdots)} \quad\text{and}\quad {\color{supp}\log_{b}}(\cdots) = \frac{{\color{supp}\ln}(\cdots)}{\color{supp}\ln(b)}.
  \]

  \bigskip
  Useful identities (if you want to memorize as little as possible):
  \begin{align*}
    \ln(e) &= 1, &\quad \ln(ab) &= \ln(a) + \ln(b),\\
    \ln(a^{x}) &= x \ln(a) &\quad \ln(a/b) &= \ln(a) - \ln(b).
  \end{align*}
\end{frame}


\section{Limits and continuity}

\begin{frame}
  \frametitle{The idea of limits}

  \faLightbulb{} To \emph{estimate} \(\lim_{x \to a} f(x)\), we move \(x\) closer and closer to the constant \(a\) (but not touching \(a\)) from both sides and look for a pattern: Does \(f(x)\) eventually get arbitrarily closer to some constant \(L\)? 

  \begin{center}
    If such an \(L\) \hlmain{exists}, then \(\lim_{x \to a} f(x) = L\); otherwise, the limit \(\lim_{x \to a}f(x)\) \hlmain{does not exist}.
  \end{center}
\end{frame}

\end{frame}

\begin{frame}
  \frametitle{Limit Laws (\(+, -, \times, \div\))}

  Suppose \(\lim_{x \to a} f(x)\) and \(\lim_{x \to a} g(x)\) \hlwarn{both exist}. Then
  \begin{enumerate}[label=(\arabic*)]
    \item \(\lim_{x \to a} [f(x) + g(x)] = \lim_{x \to a} f(x) + \lim_{x \to a} g(x)\) \hfill (sum)
    \item \(\lim_{x \to a} [f(x) - g(x)] = \lim_{x \to a} f(x) - \lim_{x \to a} g(x)\) \hfill (difference)
  \end{enumerate}

  \vspace{3ex}
  \begin{enumerate}[label=(\arabic*), start=4]
    \item \(\lim_{x \to a} [f(x) \; g(x)] = \lim_{x \to a} f(x) \cdot \lim_{x \to a} g(x)\) \hfill (product)
    \item \(\lim_{x \to a} \frac{f(x)}{{\color{warn} g(x)}} = \frac{\lim\limits_{x \to a} f(x)}{{\color{warn} \lim\limits_{x \to a} g(x)}}\), \quad \hlwarn{if {\(\lim_{x \to a}g(x) \ne 0\)}} \hfill (quotient)
  \end{enumerate}
\end{frame}


\begin{frame}
  \frametitle{Infinite limits (special cases of non-existent limit)}
  \[
    \begin{array}{rcl}
      \lim_{x \to a^{+}} \frac{1}{(x-a)^{\text{odd positive power}}} &=& + \infty  \\
      \lim_{x \to a^{{\color{warn}-}}} \frac{1}{(x-a)^{\text{odd positive power}}} &=& {\color{warn} - \infty}  \\[4ex]
      \lim_{x \to a^{-}} \frac{1}{(x-a)^{\text{even positive power}}} &=& + \infty \\
      \lim_{x \to a^{+}} \frac{1}{(x-a)^{\text{even positive power}}} &=& + \infty
    \end{array}
  \]
\end{frame}


\begin{frame}
  \frametitle{Intermediate Value Theorem}

  Let \(f\) be a \hlwarn{continuous} function over a closed, bounded interval \([a,b]\).  If \(z\) is any real number between \(f(a)\) and \(f(b)\), then \hlmain{there exists} a number \(c\) in \([a,b]\) satisfying \(f(c) = z\).

  \pause
  \bigskip

  \faLightbulb{} IVT answers the yes-no question: Is it possible to find at least one solution
  \[
    f(\text{unknown}) = (\text{given constant})
  \] 
  on the interval \([a,b]\)?
\end{frame}


\section{Derivatives}
\begin{frame}
  \frametitle{Concepts associated to \(f'\)}

  \textbf{Definition} (aka first principle):
  \[
    f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}
    \quad\text{or}\quad
    f'(a) = \lim_{x \to a} \frac{f(x) - f(a)}{x - a}.
  \]

  \textbf{Geometry}: 
  \begin{center}
    The number \(f'(a)\) is the slope of the graph of \(f(x)\) at \(x = a\).
  \end{center}


\end{frame}


\begin{frame}
  \frametitle{Differentiation Rules (1)}

  \[
    \begin{array}{rcll}
      \frac{d}{dx} \bigg( f(x) + g(x) \bigg) &=& f'(x) + g'(x) & \text{(sum)} \\[1ex] 
      \frac{d}{dx} \bigg( f(x) - g(x) \bigg) &=& f'(x) - g'(x) & \text{(difference)} \\[1ex]
      \frac{d}{dx} c f(x) &=& c f'(x) & \text{(constant multiple)} \\[1ex]
    \end{array}
  \]
\end{frame}

\begin{frame}
  \frametitle{Differentiation Rules (2)}

  \[
    \begin{array}{rcll}
      \frac{d}{dx} f(x) g(x) &=& f'(x)g(x) + f(x)g'(x) & \text{(product)} \\[1ex] 
      \frac{d}{dx}  &=& f'(x) - g'(x) & \text{(quotient)} \\[1ex]
      \frac{d}{dx} c f(x) &=& c f'(x) & \text{(constant multiple)} \\[1ex]
    \end{array}
  \]
\end{frame}

\begin{frame}[t]
  \frametitle{The Chain Rule} 

  \begin{enumerate}[itemsep=2ex]
    \item Learn WHEN we must use the chain rule.

      \begin{center}
        Whenever we differentiate a COMPOSITE function.
      \end{center}

    \item Learn HOW to apply it.
      \[
        \frac{d}{dx} \underbrace{f\bigg(\overbrace{\color{supp} g(x)}^{{\color{main}u}}\bigg)}_{y = f(u)} = \frac{dy}{d{\color{main} u}} \cdot \frac{d{\color{supp} u}}{dx} 
      \]
  \end{enumerate}

  Getting used to the Leibniz notation will help you with implicit differentiation.
\end{frame}

\begin{frame}
  \frametitle{Critical number}

  A \hlmain{critical number} \(c\) is a constant in the domain of \(f(x)\) (but NOT an endpoint) if 
  \[
    f'(c) = 0 \text{ or } f'(c) \text{ does not exist}.
  \]
  If \hlmain{\(c\) is a critical number} of \(f(x)\), then \hlsupp{\((c, f(c))\)} is called a \hlsupp{critical point}.

  \faStar{} The precise definition tells us how to find critical numbers.
\end{frame}



\begin{frame}
  \frametitle{The general strategy for finding absolute extrema}

  Let \(f(x)\) be a function defined over an interval \(I\).

  Absolute extrema can only occur at \(x\)-values \(c\) in exactly one of two ways:
  \begin{center}
    either \hlmain{\(c\) is a critical number} or \hlsupp{\(c\) is an endpoint of \(I\)}.
  \end{center}

  Note that the strategy applies to all functions.
\end{frame}

\begin{frame}
  \frametitle{Tips for algebraic tricks for l'H\^opital's rule}

  Indeterminate \(0 \cdot \infty\):
  \[
    \begin{array}{rclcrcl}
      \text{small} \cdot \text{large} &=& \frac{\text{small}}{1/\text{large}} &\longrightarrow& \underbrace{f(x)}_{\text{small}} \cdot \underbrace{g(x)}_{\text{large}} &=& \frac{f(x)}{1/g(x)}
    \end{array}
  \]

  Indeterminate \(\infty - \infty\):  

  \begin{center}
    Turn everything into a single fraction.
  \end{center}

  Indeterminate \(0^{\infty}\) or \(\infty^{0}\) or \(1^{0}\):

  \begin{center}
    Change base to \(e\) is notationally simpler.
  \end{center}

\end{frame}



\section{Integration}

\begin{frame}
  \frametitle{Antiderivatives}

  An (not the) antiderivative of \(f(x)\) is function \(F(x)\) satisfying 
  \[
    F'(x) = f(x).
  \]

  All (not an) antiderivatives of \(f(x)\) has the form \[\underbrace{F(x)}_{\text{any one antiderivative}} + C.\]
\end{frame}

\begin{frame}
  \frametitle{Definite integrals represent net area}
  \(\int_{a}^{b} f(x) \;dx\) is the \hlsupp{net area} under \(f(x)\) over the interval \([a,b]\) \hlmain{approximated by infinitely} many \hlsupp{rectangles}.

  \[
    \overbrace{\int_{a}^{b} f(x) \;dx}^{\text{\parbox{1.5in}{\centering net area under \(f(x)\)\\over \([a,b]\)}}} = \overbrace{\color{main} \lim_{n \to \infty}}^{\text{infinite}} \overbrace{\sum_{i=1}^{n} \underbrace{f(x_{i})}_{\text{\hlsupp{height}}} \underbrace{\Delta x}_{\text{\hlsupp{width}}}}^{\text{\parbox{1.5in}{\centering right-approx.\\using \(n\) rectangles}}}
  \]
\end{frame}

\begin{frame}
  \frametitle{Fundamental Theorem of Calculus (short versions)}

  \begin{enumerate}[wide, label={Part \arabic*.}, itemsep={3ex}]
    \item \(\frac{d}{dx} \int_{a}^{x} f(t) \;dt = f(x)\) as long as \(a\) is a constant.
    \item \(\int_{a}^{b} f(x) \;dx = F(b) - F(a)\) as long as \(\frac{d}{dx} F(x) = f(x)\).

      \medskip
      In part 2, we are (often) given \(f(x)\) and need to \emph{find} one antiderivative \(F(x)\).
  \end{enumerate}
\end{frame}
