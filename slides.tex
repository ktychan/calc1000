\section{Fundamentals and Reviews}
\begin{frame}
  \frametitle{Inverse functions}

  Given a function \(f\) whose domain is \(D\) and range is \(R\), its \hlmain{inverse function} (if it exists) is \hlsupp{the} function \(f^{-1}(x)\) such that
  \[
    f^{-1}(f(x)) = x \text{ on } D \quad\text{and}\quad f(f^{-1}(x)) = x \text{ on } R.
  \]

  \pause
  The two ``\(=\)'' are equalities between functions. The rules and domains on both sides of the equations must match, respectively, for the ``\(=\)'' to be valid.
\end{frame}

\begin{frame}
  \frametitle{The equivalent equation for \(f^{-1}(x)\)}

  If you know \(f(x)\) and need to work with \(y = f^{-1}(x)\), then \hlmain{the equivalent equation} \[x = f(y)\] is often useful.
\end{frame}

\begin{frame}
  \frametitle{Exponentials and logarithms}

  \large

  \[
    \begin{array}{rclcrcl}
      e^{a}e^{b} &=& e^{a + b} \\[1ex]
      (e^{a})^{b} &=& e^{ab} \\[1ex]
      \ln(ab) &=& \ln(a) + \ln(b) 
              &&
      \ln(a {\color{red}/} b) &=& \ln(a) {\,\color{red}-\,} \ln(b) \\[3ex]\pause
      b &=& e^{\ln(b)}
        &&
      b^{x} &=& e^{\ln(b)x}
    \end{array}
  \]
\end{frame}

\begin{frame}
  \frametitle{\(e^{x}\) and \(\ln(x)\) are inverses of each other and ...}
  Exponent rules: \(e^{a} e^{b} = e^{a+b}\) and \((e^{a})^{b} = e^{ab}\).

  \bigskip
  Change base to \(e\).
  \[
    {\color{main}b}^{(\cdots)} = {\color{main}e}^{{\color{main}\ln(b)}(\cdots)} \quad\text{and}\quad {\color{supp}\log_{b}}(\cdots) = \frac{{\color{supp}\ln}(\cdots)}{\color{supp}\ln(b)}.
  \]

  \bigskip
  Useful identities (if you want to memorize as little as possible):
  \begin{align*}
    \ln(e) &= 1, &\quad \ln(ab) &= \ln(a) + \ln(b),\\
    \ln(a^{x}) &= x \ln(a) &\quad \ln(a/b) &= \ln(a) - \ln(b).
  \end{align*}
\end{frame}


\section{Limits and continuity}

\begin{frame}
  \frametitle{The idea of limits}

  \faLightbulb{} To \emph{estimate} \(\lim_{x \to a} f(x)\), we move \(x\) closer and closer to the constant \(a\) (but not touching \(a\)) from both sides and look for a pattern: Does \(f(x)\) eventually get arbitrarily closer to some constant \(L\)? 

  \begin{center}
    If such an \(L\) \hlmain{exists}, then \(\lim_{x \to a} f(x) = L\); otherwise, the limit \(\lim_{x \to a}f(x)\) \hlmain{does not exist}.
  \end{center}
\end{frame}

\begin{frame}
  \frametitle{Limit Laws (\(+, -, \times, \div\))}

  Suppose \(\lim_{x \to a} f(x)\) and \(\lim_{x \to a} g(x)\) \hlwarn{both exist}. Then
  \begin{enumerate}[label=(\arabic*)]
    \item \(\lim_{x \to a} [f(x) + g(x)] = \lim_{x \to a} f(x) + \lim_{x \to a} g(x)\) \hfill (sum)
    \item \(\lim_{x \to a} [f(x) - g(x)] = \lim_{x \to a} f(x) - \lim_{x \to a} g(x)\) \hfill (difference)
  \end{enumerate}

  \vspace{3ex}
  \begin{enumerate}[label=(\arabic*), start=4]
    \item \(\lim_{x \to a} [f(x) \; g(x)] = \lim_{x \to a} f(x) \cdot \lim_{x \to a} g(x)\) \hfill (product)
    \item \(\lim_{x \to a} \frac{f(x)}{{\color{warn} g(x)}} = \frac{\lim\limits_{x \to a} f(x)}{{\color{warn} \lim\limits_{x \to a} g(x)}}\), \quad \hlwarn{if {\(\lim_{x \to a}g(x) \ne 0\)}} \hfill (quotient)
  \end{enumerate}
\end{frame}



\begin{frame}
  \frametitle{Infinite limits (special cases of non-existent limit)}
  \[
    \begin{array}{rcl}
      \lim_{x \to a^{+}} \frac{1}{(x-a)^{\text{odd positive power}}} &=& + \infty  \\
      \lim_{x \to a^{{\color{warn}-}}} \frac{1}{(x-a)^{\text{odd positive power}}} &=& {\color{warn} - \infty}  \\[4ex]
      \lim_{x \to a^{-}} \frac{1}{(x-a)^{\text{even positive power}}} &=& + \infty \\
      \lim_{x \to a^{+}} \frac{1}{(x-a)^{\text{even positive power}}} &=& + \infty
    \end{array}
  \]
\end{frame}

\begin{frame}
  \frametitle{Intermediate Value Theorem}

  \faLightbulb{} IVT answers the yes-no question: Is it possible to find at least one solution
  \[
    f(\text{unknown}) = (\text{given constant, call it } z)
  \] 
  so that the unknown lies in the interval \([a,b]\)?

  To apply IVT, check that the given constant \(z\) falls between \(f(a)\) and \(f(b)\), i.e., \(f(a) \le z \le f(b)\) or \(f(a) \ge z \ge f(b)\).
\end{frame}

\begin{frame}
  \frametitle{Limits at infinity}

  Let's say the variable in the limit is \(x\).

  Factor out \hlmain{the largest powers of \(x\)} \hlwarn{with their coefficients} in the numerator and denominator, leaving everything else with an \(x\) as \(\frac{1}{x^{\text{positive exponent}}}\).

  For limits at \(-\infty\), start by substituting \(u = -x\).
\end{frame}

\section{Derivatives}


\begin{frame}
  \frametitle{Concepts associated to \(f'\)}

  Focus on these two concepts when solving conceptual problems.

  \textbf{Definition} 
  \[
    f'(a) = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}
    \quad\text{or}\quad
    f'(a) = \lim_{x \to a} \frac{f(x) - f(a)}{x - a}
  \]

  \textbf{Geometry}:  
  The number \(f'(a)\) is the \hlmain{slope} of the tangent line \(f(x)\) at \(x = a\).  The equation of the tangent line is
  \[
    y = f'(a) (x - a) + f(a).
  \]
  
\end{frame}

\begin{frame}
  \frametitle{Pro-tips for calculating derivatives}

  Allow the Leibniz notation \(\frac{d}{dx} \cdots\) to guide you to the next step. 

  Keep your writing well organized!
\end{frame}

\begin{frame}
  \frametitle{Differentiation Rules \hfill {\footnotesize (part \(1\) of \(3\))}}
  \begin{multicols}{2}
    \begin{enumerate}[label=(\arabic*)]
      \item \(\frac{d}{dx} (\text{constant}) = 0\)
      \item \(\frac{d}{dx} x^{n} = n x^{n-1}\)
      \item \(\frac{d}{dx} e^{x} = e^{x}\)
      \item \(\frac{d}{dx} \sin(x) = \cos(x)\)
      \item \(\frac{d}{dx} \cos(x) = -\sin(x)\)
      \item \(\frac{d}{dx} b^{x} = \ln(b)b^{x}\)
    \end{enumerate}
  \end{multicols}
\end{frame}

\begin{frame}
  \frametitle{Differentiation Rules \hfill {\footnotesize (part \(2\) of \(3\))}}

  If \(f\) and \(g\) are \hlwarn{differentiable} functions, then
  \[
    \begin{array}{rcll}
      \frac{d}{dx} \bigg( f(x) + g(x) \bigg) &=& f'(x) + g'(x) \\[1ex] 
      \frac{d}{dx} \bigg( f(x) - g(x) \bigg) &=& f'(x) - g'(x) \\[1ex]
      \frac{d}{dx} (\text{constant}) f(x) &=& (\text{constant}) f'(x) \\[1ex]
      \frac{d}{dx} f(x) g(x) &=& f'(x)g(x) + f(x)g'(x) \\[1ex] 
      \frac{d}{dx} \frac{f(x)}{g(x)} &=& \frac{f'(x)g(x) - f(x)g'(x)}{\big[g(x)\big]^{2}}
    \end{array}
  \]
\end{frame}

\begin{frame}[t]
  \frametitle{Differentiation Rules \hfill {\footnotesize (part \(3\) of \(3\))}}

  \begin{enumerate}[itemsep=2ex]
    \item Learn WHEN we must use the chain rule.

      \begin{center}
        Whenever we differentiate a COMPOSITE function.
      \end{center}

    \item Learn HOW to apply it.
      \[
        \frac{d}{dx} f\bigg({\color{supp}g(x)}\bigg) = f'\Big({\color{supp}g(x)}\Big) g'(x)
        \quad\text{or}\quad
        \frac{d}{dx} \underbrace{f\bigg(\overbrace{\color{supp} g(x)}^{{\color{main}u}}\bigg)}_{y = f(u)} = \frac{dy}{d{\color{main} u}} \cdot \frac{d{\color{supp} u}}{dx} 
      \]
  \end{enumerate}

  Getting used to the Leibniz notation will help you with implicit differentiation.
\end{frame}

\begin{frame}
  \frametitle{Solving for \(x\) in \(\tfrac{f(x)}{g(x)} = 0\)}

  \begin{enumerate}[itemsep={2ex}]
    \item Multiplying through by the denominator (never zero) is always okay! In symbols, 
      \[
        \frac{f(x)}{g(x)} = 0 \quad\longrightarrow\quad f(x) = 0.
      \]

    \item Dividing through by something non-zero is always okay!

    \item Dividing through by something (may or may not be zero) POTENTIALLY throws away solutions. 

  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Logarithmic Differentiation}
  \begin{enumerate}
    \item Take \(\ln(\cdots)\) on both sides and fully simplify.
    \item Implicitly differentiate.  Use 
      \[
        \frac{d}{dx} ln\Big(g(x)\Big) = \frac{g'(x)}{g(x)}.
      \]
    \item Isolate for \(\frac{dy}{dx}\).
    \item Restore \(\frac{dy}{dx}\) as an expression of \(x\) if possible.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Critical number}

  A \hlmain{critical number} \(c\) is a constant in the domain of \(f(x)\) (but NOT an endpoint) if 
  \[
    f'(c) = 0 \text{ or } f'(c) \text{ does not exist}.
  \]
  If \hlmain{\(c\) is a critical number} of \(f(x)\), then \hlsupp{\((c, f(c))\)} is called a \hlsupp{critical point}.

  \faStar{} The precise definition tells us how to find critical numbers.
\end{frame}



\begin{frame}
  \frametitle{The general strategy for finding absolute extrema}

  Let \(f(x)\) be a function defined over an interval \(I\).

  Absolute extrema can only occur at \(x\)-values \(c\) in exactly one of two ways:
  \begin{center}
    either \hlmain{\(c\) is a critical number} or \hlsupp{\(c\) is an endpoint of \(I\)}.
  \end{center}

  Note that the strategy applies to all functions.
\end{frame}

\begin{frame}
  \frametitle{Specific strategies for finding absolute extrema}

  Always start by looking for critical numbers.
  \begin{enumerate}
    \item Compare values at endpoints and at critical numbers.  Too much work? Try the next two.
    \item Use the first derivative test.  Hard to figure out signs? Try the next one.
    \item Use the second derivative test. 
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Derivative Tests}

  \begin{itemize}
    \item \(f'>0\) on \((a,b)\) means \(f\) is increasing on \((a,b)\).
    \item \(f'<0\) on \((a,b)\) means \(f\) is decreasing on \((a,b)\).
    \item \(f''>0\) on \((a,b)\) means \(f\) is concave up on \((a,b)\).
    \item \(f''<0\) on \((a,b)\) means \(f\) is concave down on \((a,b)\).
  \end{itemize}

  At a critical number \(c\), sketch a picture according to increasing/decreasing or concavity to determine if it is a local min, local max or neither. 
\end{frame}

\begin{frame}
  \frametitle{L'H\^opital's Rule}

  For indeterminate of types \(0/0\) or \(\infty/\infty\), l'H\^opital's rule says
  \[
    \lim_{x \to a} \frac{f(x)}{g(x)} = \lim_{x \to a} \frac{f'(x)}{g'(x)}.
  \]

  Apply l'H\^opital's rule repeatedly until the limit can be fully evaluated.
\end{frame}

\begin{frame}
  \frametitle{Tips for algebraic tricks for l'H\^opital's rule}

  Indeterminate \(0 \cdot \infty\):
  \[
    \begin{array}{rclcrcl}
      \text{small} \cdot \text{large} &=& \frac{\text{small}}{1/\text{large}} &\longrightarrow& \underbrace{f(x)}_{\text{small}} \cdot \underbrace{g(x)}_{\text{large}} &=& \frac{f(x)}{1/g(x)}
    \end{array}
  \]

  Indeterminate \(\infty - \infty\):  

  \begin{center}
    Turn everything into a single fraction.
  \end{center}

  Indeterminate \(0^{\infty}\) or \(\infty^{0}\) or \(1^{0}\):

  \begin{center}
    Change base to \(e\) is notationally simpler.
  \end{center}
\end{frame}

\begin{frame}
  \frametitle{Related Rates}

  \begin{enumerate}
    \item Introduce notations. Find an equation that relates all given quantities.
    \item Make your goal clear to yourself: Find \(\frac{d}{dt} (\text{desired quantity})\) given ....
    \item Find an expression for \(\frac{d}{dt} (\text{desired quantity})\) using implicit differentiation.
    \item The expression for \(\frac{d}{dt} (\text{desired quantity})\) tells you exactly what you need to achieve your goal.
  \end{enumerate}
\end{frame}

\begin{frame}
  \frametitle{Optimization}

  \begin{enumerate}
    \item Describe the objective variable as a single-variable function.
      \begin{enumerate}
        \item Sketch the given scenario and introduce variables to measure relevant quantities.
        \item Identify the objective variable. Write ``Goal: We need to maximize/minimize ....''
        \item Find relations among our variable.
        \item Eliminate all but one independent variable.
      \end{enumerate}
    \item Maximize or minimize the objective variable.
  \end{enumerate}
\end{frame}

\section{Integration}

\begin{frame}
  \frametitle{Antiderivatives}

  If \({\color{supp} f(x)}\) and \({\color{main} F(x)}\) are related by 
  \(\frac{d}{dx} \hlmain{F(x)} = \hlsupp{f(x)}\) for every \(x\) in the domain of \(f(x)\),
  then we say \({\color{main} F(x)}\) is \emph{an} (not the) antiderivative of \({\color{supp} f(x)}\).

  \pause 
  Examples. 
  \begin{itemize}
    \item Because \(\frac{d}{dx} \hlmain{(x^{2})} = \hlsupp{2x}\), we say \pause \({\color{main} x^{2}}\) is \emph{an} antiderivative of \({\color{supp} 2x}\).
    \item \pause Because \(\frac{d}{dx} \hlmain{(x^{2} + 3)} = \hlsupp{2x}\), we say \pause \({\color{main} x^{2}} + 3\) is also \emph{an} antiderivative of \({\color{supp} 2x}\).
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Indefinite Integrals}

  All antiderivatives of \(f(x)\) has the form \[\underbrace{F(x)}_{\text{any one antiderivative}} + C.\]
\end{frame}

\begin{frame}
  \frametitle{Riemann Sum}
  \phantom{\(\int_{a}^{b} f(x) \;dx\) is the \hlsupp{net area} ``under'' \(f(x)\) over the interval \([a,b]\) \hlmain{approximated by infinitely} many \hlsupp{rectangles}.}

  \[
    \phantom{\overbrace{\int_{a}^{b} f(x) \;dx}^{\text{\parbox{1.5in}{\centering net area ``under'' \(f(x)\)\\over \([a,b]\)}}} =} \overbrace{\color{main} \lim_{n \to \infty}}^{\text{infinite}} \overbrace{\sum_{i=1}^{n} \underbrace{f(x_{i}^{*})}_{\text{\hlsupp{height}}} \underbrace{\Delta x}_{\text{\hlsupp{width}}}}^{\text{\parbox{1.5in}{\centering approximation \\using \(n\) rectangles}}}
  \]
\end{frame}

\begin{frame}
  \frametitle{\onslide<2->{Definite integrals represent net area}}
  \onslide<2->{
  \(\int_{a}^{b} f(x) \;dx\) is the \hlsupp{net area} ``under'' \(f(x)\) from \(a\) to \(b\).}

  \[
    \overbrace{\int_{a}^{b} f(x) \;dx}^{\text{\parbox{1.5in}{\centering net area ``under'' \(f(x)\)\\over \([a,b]\)}}} = \overbrace{\color{main} \lim_{n \to \infty}}^{\text{infinite}} \overbrace{\sum_{i=1}^{n} \underbrace{f(x_{i}^{*})}_{\text{\hlsupp{height}}} \underbrace{\Delta x}_{\text{\hlsupp{width}}}}^{\text{\parbox{1.5in}{\centering right-approx.\\using \(n\) rectangles}}}
  \]

  Choosing \(x_{i}^{*}\) to be the right endpoints yields notationally simpler expressions.
\end{frame}

\begin{frame}
  \frametitle{Not so obvious properties of integrals}

  \begin{align*}
    \int_{a}^{b} f(x) \;dx &= - \int_{b}^{a} f(x) \;dx &&\text{(flip)}\\
    \int_{a}^{b} f(x) \;dx &= \int_{a}^{c} f(x) \;dx + \int_{c}^{b} f(x) \;dx &&\text{(split/merge)}.
  \end{align*}

  All properties can be understood conceptually by drawing pictures.
\end{frame}

\begin{frame}
  \frametitle{Fundamental Theorem of Calculus (short versions)}

  Suppose \(f(x)\) is continuous.

  \begin{align*}
    \frac{d}{d{\color{main}x}} \int_{\text{constant}}^{{\color{main}x}} {\color{main}f}(t) \;dt &= {\color{main}f(x)} && \text{(part 1)} \\[2ex] 
    \onslide<2->{{\color{main} \int_{{\color{supp}a}}^{{\color{supp}b}} f(x) \;dx} &= \left[{\color{main}\int f(x) \;dx}\right]_{{\color{supp}a}}^{{\color{supp}b}} && \text{(part 2)}}
  \end{align*}
\end{frame}


\begin{frame}
  \frametitle{Substitution for integrals}

  Substitute \({\color{main}u = g(x)}\) and \(\color{supp} g'(x)\;dx = du\). 
  Remember that \(g'\) is \emph{calculated} from the \emph{choice} for \(g\).

  \begin{align*}
    \int f({\color{main}g(x)}) {\color{supp} g'(x) dx} & \onslide<2->{\underbrace{= \int f({\color{main} u}) \; {\color{supp}du}}_{\text{substitution}}} \\[2ex]
    \int_{a}^{b} f({\color{main}g(x)}) {\color{supp} g'(x) dx} &\onslide<3->{\underbrace{= \int_{u(a)}^{u(b)} f({\color{main} u}) \; {\color{supp}du}}_{\text{substitution}}} \onslide<4->{\underbrace{= \left[ \int f({\color{main} u}) \; {\color{supp}du} \right]_{u(a)}^{u(b)}}_{\text{FTC part 2}}}
  \end{align*}
\end{frame}

\begin{frame}
  \frametitle{Area between curves}

  If integrating with respect to \(x\), then
  \[
    \text{area} = \int_{\text{left}}^{\text{right}}  (\text{upper}) - (\text{lower})  \; dx
  \]

  If integrating with respect to \(y\), then
  \[
    \text{area} = \int_{\text{bottom}}^{\text{top}}  (\text{right}) - (\text{left})  \; dy
  \]
\end{frame}

\begin{frame}
  \frametitle{Volume of solids of revolution}

  If the solid is obtained by rotating a region about the \(x\)-axis, then
  \[
    \text{volume} = \int_{\text{left}}^{\text{right}} \pi \left( \text{\parbox{1.5cm}{\centering outer\\radius}} \right)^{2} - \pi \left( \parbox{1.5cm}{\centering inner\\radius} \right)^2 \;dx
  \]

 
  \pause
  If the solid is obtained by rotating a region about the \(y\)-axis, then
  \[
    \text{volume} = \int_{\text{bottom}}^{\text{top}} \pi \left( \text{\parbox{1.5cm}{\centering outer\\radius}} \right)^{2} - \pi \left( \parbox{1.5cm}{\centering inner\\radius} \right)^2 \;dy
  \]

  Solve for the points of intersections to determine the upper/lower or left/right bounds.
\end{frame}
