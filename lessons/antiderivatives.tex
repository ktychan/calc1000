\begin{definition}
  If \(f(x)\) and \(F(x)\) are related by
  \[
    f(x) = F'(x) \text{ for every } x \text{ in the domain of } f(x),
  \]
  then we \emph{say} \(F(x)\) is \emph{an} antiderivative of \(f(x)\).
\end{definition}

Let's make sense of the above definition.

\end{example}

\begin{definition} \label{def:indefinite-integral}
The \hlmain{indefinite integral} of \(f\) is a single expression representing all antiderivatives of \(f(x)\) and is denoted by \({\color{main} \int f(x) \;dx}\). If \(F(x)\) is \emph{an} antiderivative of \(f(x)\), then 
  \[
    \int f(x) \;dx = F(x) + (\text{constant}),
  \] 
  or more conventionally, \( \int f(x) \;dx = F(x) + C\) where \(C\) represents a general constant.

  \textbook{Definition, page 487}
\end{definition}

\begin{example}
  Definition~\ref{def:indefinite-integral} help us evaluate \emph{simple} integrals by recognizing that the integrand is the derivative of \emph{something}. 

  \begin{multicols}{2}
    \begin{enumerate}[itemsep={2cm}]
      \item \(\int 0 \;dx = \phantom{\text{(any constant)} + C}  \)
      \item \(\int e^{x}\;dx = \phantom{e^{x} + C}\)
      \item \(\int 10 \;dx = \phantom{10x + C}\)
      \item \(\int \sin(x) \;dx = \)
      \item \(\int 4x^{3} \;dx = \)
      \item \(\int x^{5} \;dx = \)
      \item \(\int \sqrt{x} \;dx = \)
      \item \(\int \frac{1}{\sqrt{x}} \;dx = \)
      \item \(\int \frac{1}{1 + x^{2}} \;dx = \)
      \item \(\int \cos(x) \;dx = \)
      \item \(\int \sec^{2}(x) \;dx = \)
      \item \(\int \tan(x)\sec(x) \;dx = \)
    \end{enumerate}
  \end{multicols}
  
\end{example}

Algebraic properties of integrals break up complex integrals into simple ones.

\begin{thm}
  \begin{align*}
    \int c f(x) \;dx 
    &= c \int f(x) \;dx && \text{(constant multiple)}\\
    \int f(x) + g(x) \;dx 
    &= \int f(x) \;dx + \int g(x) \;dx && \text{(sum)}\\
    \int f(x) - g(x) \;dx 
    &= \int f(x) \;dx - \int g(x) \;dx && \text{(difference)}
  \end{align*}

  \textbook{Theorem~4.16, page 491}
\end{thm}
