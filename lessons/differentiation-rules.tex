\begin{document} \section{Differentiation rules (with respect to \texorpdfstring{\(+,-,\times,\div\)}{+, -, •, ÷})}

\textbf{Notations}. We write \(y = f(x)\) to denote a function. All of the following denote the derivative of \(f(x)\) at a constant \(a\).
\begin{align*}
  & f'(a) \quad=\quad \frac{dy}{dx} \bigg|_{x = a} \quad=\quad \frac{dy}{dx} \bigg]_{x = a}
\end{align*}


\emph{All} of the following denote the derivative of \(f(x)\) as a function.
\[
  f'(x) 
  \;=\; f' 
  \;=\; y' 
  \qquad =\qquad  
  {\color{main} \frac{d}{dx}} \, f(x) \;=\; \frac{df}{dx} \;=\; \frac{dy}{dx}
  % \qquad=\quad 
  % Df(x) 
  % \;=\; D_{x} f(x)
\]

\label{page:higher-derivatives}

In general, we define \(f^{(0)} = f\) and the \(n\)-th derivative to be
\[
  \overbrace{f^{' \cdots '}}^{\text{\(n\) primes}} \quad=\quad f^{(n)} \quad=\quad \frac{d^{n}}{dx^{n}} f \quad=\quad \frac{d}{dx} f^{(n-1)} \quad\text{for integers } n \ge 1.
\]

\begin{example}
  \(f''(x) = f^{(2)}(x) = \frac{d}{dx} \left( \tfrac{d}{dx} f(x) \right)\) and \(f^{(3)}(x) = \frac{d}{dx}\left( \frac{d}{dx} \left( \tfrac{d}{dx} f(x) \right) \right)\) and so on.
\end{example}

\bigskip
Differentiation rules are patterns of calculations and express the interactions between differentiation (as an operation) and algebraic operations \(+, -, \times, \div, \circ\). 

\begin{mdframed}[style=simple-compact] \label{thm:diff-rules-1}
  \textbf{Differentiation Rules} \hfill {\footnotesize (part \(1\) of \(3\))}
  \begin{multicols}{2}
    \begin{enumerate}[label=(\arabic*)]
      \item \(\frac{d}{dx} (\text{constant}) = 0\)
      \item \(\frac{d}{dx} x^{n} = n x^{n-1}\)
      \item \(\frac{d}{dx} e^{x} = e^{x}\)
      \item \(\frac{d}{dx} \sin(x) = \cos(x)\)
      \item \(\frac{d}{dx} \cos(x) = -\sin(x)\)
      \item \(\frac{d}{dx} b^{x} = \ln(b)b^{x}\)
    \end{enumerate}
  \end{multicols}
\end{mdframed}

\begin{mdframed}[style=simple]
  \textbf{Differentiation Rules} \hfill {\footnotesize (part \(2\) of \(3\))}

  If \(f\) and \(g\) are \hlwarn{differentiable} functions, then
  \begin{align*}
    \frac{d}{dx} f(x) + g(x) &= \frac{d}{dx}f(x) + \frac{d}{dx}g(x) && \text{(sum)} \\[1ex]
    \frac{d}{dx} f(x) - g(x) &= \frac{d}{dx}f(x) - \frac{d}{dx}g(x) && \text{(difference)} \\[1ex]
    \frac{d}{dx} (\text{constant}) f(x) &= (\text{constant}) \, \frac{d}{dx} f(x),  && \text{(constant multiple)} \\
    \frac{d}{dx} f(x)g(x) &= { f'(x) g(x) + f(x) g'(x) } && \text{(product)} \\[1ex]
    \frac{d}{dx} \frac{f(x)}{g(x)} &= {\frac{f'(x)g(x) - f(x)g'(x)}{ \big[g(x)\big]^{2} }} && \text{(quotient)}
  \end{align*}
\end{mdframed}

Derivatives give us equations, which means that we can solve questions in the form ``find all values of \ldots{} for which the derivative of \ldots{} satisfies \ldots{}''

\begin{example}
  Let \(f(x) = ax^{3} + bx^{2}\). Find \(a\) and \(b\) so that \(f'(1) = 3\) and \(f'(-1) = 2\).

  In other words, find all values of \(a\) and \(b\) for which the derivative of \(f(x) = ax^{3} + bx^{2}\) satisfies \(f'(1) = 3\) and \(f'(-1) = 2\).

  \blanklines{25}
\end{example}
