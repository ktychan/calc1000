\begin{exercise}
  Evaluate the following integrals.

  \begin{multicols}{3}
    \begin{enumerate}[wide, label=(\alph*)]
      \item \(\int \frac{1}{\sqrt{1 - x}} \;dx\)
      \item \(\int \frac{1}{\sqrt{1 - x^{2}}} \;dx\)
      \item \(\int \frac{\sec(x)\tan(x)}{\sqrt{\sec^{2}(x) - 1}} \;dx\) \updatenote{The radicand had \(1\) and \(\sec^{2}(x)\) swapped: \(1 - \sec^{2}(x) \le 0\) always (since \(\sec^{2}(x) \ge 1\)), so the square root was never real. Swapping to \(\sec^{2}(x) - 1 = \tan^{2}(x)\) (Pythagorean identity) makes the integral real-valued and solvable.}
    \end{enumerate}
  \end{multicols}

  \blanklines{45}
\end{exercise}
\clearpage
