\begin{exercise} \label{ex:choose-substitution}
  Evaluate the following integrals.

  \begin{multicols}{3}
    \begin{enumerate}[wide, label=(\alph*)]
      % \item \(\int_{-1}^{2} x(x^{2} + 1) \;dx\)
      \item \(\int_{-1}^{0} \frac{1}{x^{2} + 1} \;dx\)
      \item \(\int \frac{x^{2} + 1}{x} \;dx\)
      \item \(\int_{2}^{0} \frac{x}{x^{2} + 1} \;dx\)
    \end{enumerate}
  \end{multicols}

  \blanklines{50}
\end{exercise}

\begin{exercise}
  Write down the numerical values of the following integrals.  Note that numerical values should be exact, not decimal approximations.

  \begin{multicols}{3}
    \begin{enumerate}[wide, label=(\alph*)]
      \item \(\int_{2\pi}^{\pi} \tan(x)\sec(x) \;dx\)
      \item \(\int_{0}^{-\pi/4} \cos(x)\sin(x) \;dx\)
      \item \(\int_{0}^{\pi/4} \cos(\sin(2x))\cos(2x) \;dx\)
    \end{enumerate}
  \end{multicols}

  \blanklines{45}
\end{exercise}

\begin{exercise}
  Express the volume of a unit sphere obtained by rotating the semicircle \(y = \sqrt{1 - x^{2}}\) about the \(x\)-axis as an integral.

  Repeat the exercise by expressing the volume of a unit sphere obtained by rotating the ``sideways'' semicircle \(x = \sqrt{1 - y^{2}}\) as an integral.

  \blanklines{50}
\end{exercise}
