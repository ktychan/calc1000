\begin{document} \section{Algebraic techniques for evaluating limits}
Direct substitution and limit laws are the main algebraic tools for evaluating limits. 

  % \faExclamationTriangle{} This section covers a bit more than the corresponding textbook section and is organized differently to minimize redundancy. See the last page for a short comparison.

\begin{thm}[direct sub]
  If \(f(x)\) is a polynomial or a rational function, then
  \[
    \lim_{x \to a} f(x) = f(a),
  \]
  but \(a\) \hlwarn{must be} in the domain of \(f(x)\) and not an infinity.

  \textbook{Theorem 2.6 on page 162}
\end{thm}
% Note that a must be in the domain of f and is not an infinity.
% DO NOT DO arithmetic with infinity.  

% \faExclamationTriangle{} If your intuition from high school leads to doing arithmetic with infinities, then that's a good sign that you need to use limit laws explicitly.
\begin{thm}[limit laws] \label{thm:limit-laws}
  Suppose \(\lim_{x \to a} f(x)\) and \(\lim_{x \to a} g(x)\) \hlwarn{both exist}. Then
  \begin{enumerate}[label=(\arabic*)]
    \item \(\lim_{x \to a} [f(x) + g(x)] = \lim_{x \to a} f(x) + \lim_{x \to a} g(x)\) \hfill (sum)
    \item \(\lim_{x \to a} [f(x) - g(x)] = \lim_{x \to a} f(x) - \lim_{x \to a} g(x)\) \hfill (difference)
    \item \(\lim_{x \to a} [c \; f(x)] = c \lim_{x \to a} f(x)\), where \(c\) is a constant \hfill (constant multiple)
    \item \(\lim_{x \to a} [f(x) \; g(x)] = \lim_{x \to a} f(x) \cdot \lim_{x \to a} g(x)\) \hfill (product)
    \item \(\lim_{x \to a} \frac{f(x)}{{\color{warn} g(x)}} = \frac{\lim\limits_{x \to a} f(x)}{{\color{warn} \lim\limits_{x \to a} g(x)}}\), \quad \hlwarn{if {\(\lim_{x \to a}g(x) \ne 0\)}} \hfill (quotient)
    \item \(\lim_{x \to a} \bigg[f(x)\bigg]^{n} = \bigg[ \lim_{x \to a} f(x) \bigg]^{n}\) \hfill (power)
    \item \(\lim_{x \to a} \sqrt[n]{f(x)} = \sqrt[n]{\lim_{x \to a} f(x)}\), \quad if \(n\) is even, we assume {\(\lim_{x \to a} f(x) \ge 0\)} \hfill (root)
  \end{enumerate}

  \textbook{Theorem 2.5 on page 161}
\end{thm}

\faStar{} All limit laws apply to \hlmain{one-sided} (of the same type) as well as \hlmain{two-sided} limits.

Limit laws are useful when only abstract information is available.
\begin{example} \label{eg:limit-law-abstract}
  Suppose \(\lim_{x \to -3} f(x) = 4\). Evaluate \(\lim_{x \to -3} \left( f(x) \right)^{3/2}\) if possible.

  \blanklines{10}
\end{example}

Limit laws also apply to one-sided limits and piecewise functions.
\begin{example}
  Let \(f(x) = \begin{cases}\sqrt{x - 1} + 1, &x > 1\\[1ex]\frac{2x}{x + 1}, &x < 1\end{cases}\). Evaluate \(\lim_{x \to 1} f(x)\).

  \blanklines{15}
\end{example}

We turn to situations for which the quotient rule does not apply directly, i.e., when the limit of the denominator is \(0\).  If the limit of the numerator is a constant and the limit of the denominator is \(0\), then the entire limit is either \(+\infty\) or \(-\infty\) or does not exist (because its two one-sided limits are different infinities).

\begin{example} \label{eg:limit-constant-over-0}
  Evaluate \(\lim_{x \to 1} \frac{x}{x - 1}\).

  \blanklines{10}
\end{example}

\clearpage
If the limits of the numerator and the denominator are both \(0\), then we will need some algebra skills. In such a case, the application of the quotient rule must wait until after we manipulate the function so that the limit of the denominator is not zero.

\begin{example} \label{eg:limit-cancellation}
  Evaluate \(\lim_{t \to 2} \frac{t^{2} - 3t + 2}{t - 2}\).

  \blanklines{20}
\end{example}

\begin{example} \label{eg:limit-rationalization}
  Evaluate \(\lim_{x \to 0} \frac{\sqrt{x^{2}+1}-1}{x^{2}}\).

  \blanklines{20}
\end{example}

\begin{thm}[Squeeze] \label{thm:limit-squeeze}
  Let {\({\color{main}f(x)} \le g(x) \le {\color{supp}h(x)}\)} when \(x\) is near \(a\) and 
  \[ 
    \lim_{x \to a} {\color{main}f(x)} = L \quad\text{and}\quad \lim_{x \to a} {\color{supp}h(x)} = L,
  \]
  then
  {\(\lim_{x \to a} g(x) = L\).}

  \textbook{Theorem 2.7 on page 171}
\end{thm}


\begin{example} \label{ex:limit-law-x-square-sin(x)}
  Evaluate \(\lim_{x \to 0} x^{2} \sin\left(\frac{1}{x}\right)\).

  \blanklines{40}
\end{example}

\begin{exercise} \label{ex:limit-law-x-sin(x)}
  Evaluate \(\lim_{x \to 0} x \sin\left(\frac{1}{x}\right)\).

  \medskip
  \begin{minipage}{2in}
    \includestandalone[width=2in]{../standalones/plot_squeeze}
  \end{minipage}
  \hspace{0.5in}
  \begin{minipage}{4.5in}
    This problem is only a tad different from Example~\ref{ex:limit-law-x-square-sin(x)}. However, its solution is a bit more complicated. There two approaches: 
    \begin{enumerate}[{label=(\Alph*)}, itemsep={1ex}, topsep={1ex}]
      \item Use piecewise functions. 
      \item Evaluate two one-sided limits if you rather not to deal with piecewise functions.
    \end{enumerate}

    We need the squeeze theorem in both approaches.  
  \end{minipage}
\end{exercise}


\textbf{Approach (A).}
\begin{enumerate}[wide]
  \item The form of \(x \sin(1/x)\) suggests that we choose
    \begin{itemize}
      \item the lower bound to be \(f(x) = \underline{\hspace{1in}}\),
      \item the upper bound to be \(h(x) = \underline{\hspace{1in}}\).
    \end{itemize}

  \item Sketch your choices of \(f(x)\) and \(h(x)\) on the graph above. Do they \enquote{squeeze} \(x \sin(1/x)\)? If so, move on to the next step; otherwise, go back to the previous step and try again. Hint: \(f(x) = -x\) and \(h(x) = x\) are the wrong choices.

  \item The piecewise nature of \(f(x)\) and \(g(x)\) forces us to show \(f(x) \le x \sin(1/x) \le g(x)\) in two steps. 
    \begin{enumerate}
      \item Show that \(f(x) \le x \sin\left(\frac{1}{x}\right) \le h(x)\) \hlwarn{for \(x \ge 0\)}.

        \blanklines{10}

      \item Show that \(f(x) \le x \sin\left(\frac{1}{x}\right) \le h(x)\) \hlwarn{for \(x < 0\)}.

        \blanklines{10}
    \end{enumerate}

  \item Apply the Squeeze Theorem to finish evaluating the limit.

    \blanklines{4}
\end{enumerate}
\clearpage

\includestandalone[width=2in]{../standalones/plot_squeeze}

\textbf{Approach (B).} To avoid dealing with piecewise functions, we can calculate two one-sided limits.

\begin{enumerate}[wide]
  \item Use the Squeeze Theorem to evaluate \(\lim_{x \to 0^{-}} x \sin(1/x)\).

    \blanklines{20}

  \item Use the Squeeze Theorem to evaluate \(\lim_{x \to 0^{-}} x \sin(1/x)\).

    \blanklines{20}

  \item The two one-sided limits should have the same value. Conclude that \(\lim_{x \to 0} x \sin(1/x) = \underline{\hspace{1cm}}\).
\end{enumerate}

Example~\ref{ex:limit-textbook-2.28} is a subtle variation on the theme of Examples~\ref{eg:limit-cancellation} and \ref{eg:limit-rationalization} but challenges our algebra skills. It is a good practice, but leave it until you have mastered all other skills. Hopefully, the hint at the bottom of the page helps.
\begin{exercise}[Textbook Example~2.20] \label{ex:limit-textbook-2.28}
  Evaluate \(\lim_{x \to 0} \left( \frac{1}{x} + \frac{5}{x(x-5)}\right)\) or explain why it does not exist.

  \blanklines{20}
\end{exercise}
