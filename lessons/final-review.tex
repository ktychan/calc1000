\begin{exercise}
  Suppose \(f(x)\) is defined and differentiable everywhere.  Suppose \(f'(5) = 0\), \(f'\) is increasing for \(x < 5\), and decreasing for \(x > 5\).  Which statement(s) \uline{must} be true?

  \begin{enumerate}[label=\Alph*.]
    \item \(f(x)\) has at least one local extremum.
    \item \(f(x)\) has an inflection point at \(x = 5\).
    \item \(f(x)\) is increasing for \(x < 5\).
    \item \(f(x)\) is decreasing for \(x < 5\). 
    \item \(f(x)\) is concave down everywhere.
  \end{enumerate}
\end{exercise}

\vfill{}


\begin{exercise}
  Suppose \(f''(2) = 0\) and \(f''(x) \le 0\) everywhere and \(f'(3) = 0\). Choose \uline{all} true statements.

  \begin{enumerate}[label=\Alph*.]
    \item \(f(x)\) has an inflection point at \(x = 2\).
    \item \(f(x)\) has no inflection point.
    \item \(f(x)\) is concave up everywhere.
    \item \(f(x)\) is concave down everywhere.
    \item \(f(x)\) has a local minimum at \(x = 3\).
  \end{enumerate}
\end{exercise}

\vfill{}

\begin{exercise}
  Suppose \(f'(x) = \int_{e}^{x} \ln(t) \;dt\).  Which statement(s) \uline{are} true?

  \begin{enumerate}[label=\Alph*.]
    \item \(f(x)\) is increasing over the interval \((1,e)\).
    \item \(f(x)\) is decreasing over the interval \((1,e)\).
    \item \(f(x)\) is concave down for \(x > e\).
    \item \(f(x)\) is concave up for \(1 < x < e\).
    \item \(f(x)\) has no inflection point anywhere.
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Suppose \(f(x) = \begin{cases} e^{x}, &\text{ if } x < 0 \\ h(x),&\text{ if } x \ge 0. \end{cases}\) is differentiable at \(x = 0\). Which statement(s) \uline{are} true?

  \begin{enumerate}[label=\Alph*.]
    \item \(h'(x) = 0\).
    \item \(h'(x) = 1\).
    \item \(h(x) = 0\).
    \item \(h(x) = 1\).
    \item \(h(x)\) could be \(x+1\).
  \end{enumerate}
\end{exercise}
\vfill{}


\begin{exercise}
  Suppose \(F'(x) = \frac{x}{\sqrt{1+x^{2}}}\) and \(F(1) = 0\). Find \(F(x)\).

  \begin{enumerate}[label=\Alph*.]
    \item \(F(x) = \sqrt{\arctan(x)} - \sqrt{\pi/4}\).
    \item \(F(x) = \arcsin(x) - \pi/2\).
    \item \(F(x) = \sqrt{x^{2} + 1} - \sqrt{2}\).
    \item \(F(x) = \ln\left|\sqrt{x^{2} + 1}\right| - \ln(\sqrt{2})\).
    \item \(F(x) = \sqrt{x^{2} + 1}\).
  \end{enumerate}
\end{exercise}

\vfill{}

\begin{exercise}
  Differentiate \(\int_{1}^{3^{x}} \sin(t^{2})\;dt \).

  \begin{enumerate}[label=\Alph*.]
    \item \(\sin(x^{2})3^{x}\)
    \item \(\cos(x^{2})\ln(3)3^{x}\)
    \item \(\sin(3^{2x})\ln(3)3^{x}\)
    \item \(\cos(3^{2x})3^{x}\)
    \item \(\sin(3^{2x})3^{x}\)
  \end{enumerate}
\end{exercise}

\begin{exercise}
  Evaluate \(\lim_{x \to \infty} \frac{\log_{2}(2x + 1)}{\log_{3}(x)-1}\).

  \begin{enumerate}[label=\Alph*.]
    \item \(\frac{\ln(3)}{\ln(2)}\)
    \item \(\frac{\ln(3/2)}{\ln(2)}\).
    \item \(\frac{\ln(2)}{\ln(3)}\).
    \item \(\frac{\ln(2)}{\ln(3/2)}\).
    \item \(\infty\).
  \end{enumerate}
\end{exercise}
