%! TeX program = lualatex
% The rectangles for the Sigma notation example in
% lessons/the-definite-integral.tex.  Six rectangles of width 1/2 sit on
% [0, 3]; the one starting at x = i/2 has height (i/2)^2, so the total area is
%
%   A = sum_{i=0}^{5} (i/2)^2 * 1/2 = (1/8) * (0 + 1 + 4 + 9 + 16 + 25) = 55/8.
%
% The heights used to read (a + i/2)^2 for a symbolic a, over i = -3 .. 2.
% Setting a = 0 there would have put a zero height in the middle of the row,
% leaving a hole between two rectangles; starting the row at x = 0 instead puts
% the flat rectangle at the left end, where a left-endpoint staircase for x^2
% is expected to start.  Heights are drawn exactly as labelled -- the old
% picture added 1/2 to every height, which was invisible only while a stayed
% symbolic.
%
% yscale=0.6 keeps the tallest rectangle (25/4) from making the picture tall;
% at that scale the 0^2 and (1/2)^2 labels still clear each other.
\documentclass{standalone}

\input{../typography.tex.preamble}
\input{../tikz.tex.preamble}

\begin{document}
\begin{tikzpicture}[xscale=2.2, yscale=0.6]
  \draw[->] (-0.6,0) -- (3.6,0) node[right] {\footnotesize \(x\)};

  \foreach \x in {0, 1, 2, 3} {
    \node[below] at (\x, -0.05) {\footnotesize \(\x\)};
  }

  \foreach \x in {0, 0.5, 1, 1.5, 2, 2.5, 3} {
    \draw (\x, 0.1) -- (\x, -0.1);
  }

  \foreach \x in {0, 0.5, 1, 1.5, 2, 2.5} {
    \draw[draw=black!45, thin, fill=main, fill opacity=0.1]
      (\x, 0) rectangle ({\x + 0.5}, {\x*\x});
  }

  \node[above] at (0.25, 0)    {\footnotesize \(0^{2}\)};
  \node[above] at (0.75, 0.25) {\footnotesize \(\left(\frac{1}{2}\right)^{2}\)};
  \node[above] at (1.25, 1)    {\footnotesize \(1^{2}\)};
  \node[above] at (1.75, 2.25) {\footnotesize \(\left(\frac{3}{2}\right)^{2}\)};
  \node[above] at (2.25, 4)    {\footnotesize \(2^{2}\)};
  \node[above] at (2.75, 6.25) {\footnotesize \(\left(\frac{5}{2}\right)^{2}\)};
\end{tikzpicture}
\end{document}
