%! TeX program = lualatex
% Blank grids for the two Riemann-sketching exercises in
% lessons/the-definite-integral.tex, the pair that follows the worked example
% on plot_riemann_sum_draw_rectangles.
%
%   page 1 -- f(x) = 0.75x^3 - 1.25x^2 - 2x + 1.7 over [-1, 2], for n = 3
%   page 2 -- f(x) = 2 sin(0.8x + 1.3) over [0, 5], for n = 5
%
% Both functions cross the axis inside the interval, which is the whole point
% of the pair: the example's cubic is lifted clear of the axis, so every
% rectangle there sits above it, whereas here the last rectangles hang below
% and have to be counted negative.  Neither curve is zero at a subdivision
% point or a midpoint, so no rectangle a student is asked to draw collapses.
%
% Page 1 is drawn twice in the lesson, so it has to be about half of
% \textwidth (505.89pt) wide including its tick labels: 1.5cm cells over a
% 4-unit window come to roughly 185pt.  Page 2 was sized for three across at
% 0.75cm.  Include the pages at natural size -- no width=, and no scale= in
% \includestandalone -- or the cells stop being square and the curve thins.
%
% Page 1, over [-1, 2] with n = 3, puts a subdivision point on every integer:
%   left endpoints  -1, 0, 1   heights  1.7, 1.7, -0.8
%   right endpoints  0, 1, 2   heights  1.7, -0.8, -1.3
%   midpoints  -0.5, 0.5, 1.5  heights  2.29, 0.48, -1.58
% so L_3 = 2.6, R_3 = -0.4 and M_3 = 1.19 are three visibly different answers,
% and the right-endpoint one comes out negative while the left-endpoint one
% does not.  The constant term is 1.7 rather than 2 so the curve crosses the
% axis near x = 0.7: at 2 it crossed at 0.78 and grazed the "1" tick label.
%
% The same page also has to carry n = 6, where the subdivisions fall on the
% half units.  Those get minor ticks and a minor grid line each, in the same
% grey as the major ones, so a student can rule the six strips straight off
% the picture; the minor y ticks keep the cells square.  Heights there are
%   -1, -0.5, 0, 0.5, 1, 1.5, 2  ->  1.7, 2.29, 1.7, 0.48, -0.8, -1.58, -1.3
% giving L_6 = 1.895 and R_6 = 1.295.
%
% Page 2, over [0, 5] with n = 5, again divides at the integers.  The curve is
% positive up to about x = 2.3 and negative after it, so two rectangles and a
% bit stand above the axis and the rest hang below, whichever method is used.
% The phase is 1.3 rather than 1 so that the crossing lands between the ticks
% at 2 and 3: at phase 1 the curve cut straight through the "3" tick label.
\documentclass[tikz, crop]{standalone}

\input{../typography.tex.preamble}
\input{../tikz.tex.preamble}

% Matches plot_riemann_sum_draw_rectangles: the same axis lines, grid, labels
% and curve weight, so the two exercises read as the example done again.
\pgfplotsset{riemann grid/.style={
    axis lines=middle,
    axis equal image=false,
    xlabel={\(x\)}, xlabel style={anchor=west},
    ylabel={\(y\)}, ylabel style={anchor=south},
    label style={at={(ticklabel* cs:1), font=\footnotesize}},
    grid=both, grid style={gray!20},
    smooth, samples=100, no markers,
    clip=false,
  }}

\begin{document}

% ---------------------------------------------------------------------
% Page 1 -- n = 3.
\begin{tikzpicture}
  \begin{axis}[
    riemann grid,
    x=1.5cm, y=1.5cm,
    xmin=-1.5, xmax=2.5, ymin=-2.0, ymax=2.7,
    % the grid line at y = 2 is kept but its label dropped, the way the
    % example figure drops some of its own: the curve falls through y = 2 at
    % about x = -0.2 and ran straight over a label there
    xtick={-1,1,2}, ytick={-1,1,2}, yticklabels={\(-1\),\(1\),},
    minor xtick={-0.5,0.5,1.5}, minor ytick={-1.5,-0.5,0.5,1.5,2.5},
    minor grid style={gray!20},
    ]
    \addplot[main, ultra thick, domain=-1:2]
      {0.75*x^3 - 1.25*x^2 - 2*x + 1.7};
    \node[main, anchor=west] at (axis cs:2,-1.3) {\footnotesize \(f(x)\)};
  \end{axis}
\end{tikzpicture}

% ---------------------------------------------------------------------
% Page 2 -- n = 5.
\begin{tikzpicture}
  \begin{axis}[
    riemann grid,
    x=0.75cm, y=0.75cm,
    xmin=-0.5, xmax=5.5, ymin=-2.5, ymax=2.5,
    xtick={1,2,3,4,5}, ytick={-2,-1,1,2},
    ]
    \addplot[main, ultra thick, domain=0:5] {2*sin(0.8*x + 1.3)};
    \node[main, anchor=west] at (axis cs:5,-1.66) {\footnotesize \(f(x)\)};
  \end{axis}
\end{tikzpicture}

\end{document}
