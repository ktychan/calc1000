% The graph of f for the net area examples in
% lessons/the-definite-integral.tex.  f is the line 2 - x on [-2, 4], then the
% upper half of the circle of radius 1 centred at (5, -2) on [4, 6], then the
% constant 2 on [6, 10], so every piece has an area formula:
%
%   [-2, 2]  -- a triangle above the axis, area 8
%   [2, 4]   -- a triangle below the axis, area 2
%   [4, 6]   -- below the axis, a 2-by-2 square less a half disk of radius 1,
%               area 4 - pi/2
%   [6, 10]  -- a 4-by-2 rectangle above the axis, area 8
%
% So the integral over [-2, 10] is 8 - 2 - (4 - pi/2) + 8 = 10 + pi/2, and the
% total area is 8 + 2 + (4 - pi/2) + 8 = 22 - pi/2.  The two integrals in the
% example are 5/2 over [-1, 4] and 4 + pi/2 over [0, 10].
%
% The semicircle keeps both ends at height -2, and it peaks at -1, so f stays
% below the axis on [2, 6].  f jumps at x = 6, from -2 up to 2; net area does
% not care about the value at a single point, so no endpoint markers are drawn.
% basic curve keeps equal units, so the semicircle is drawn round -- the width
% went from 3in to 4.5in when the domain grew from 8 units to 12, to stop equal
% units squashing the picture vertically.
\documentclass[tikz, crop]{standalone}

\input{../typography.tex.preamble}
\input{../tikz.tex.preamble}

\begin{document}
\begin{tikzpicture}
  \begin{axis}[
    basic curve,
    width=4.5in,
    xtick={-2,-1,...,10},
    ytick={-2,-1,...,4},
    enlargelimits,
    title={The graph of \(f(x)\)},
    ]
    \addplot[very thick, black] coordinates {(-2, 4) (4,-2)};
    % parametrised rather than written as -2 + sqrt(1 - (x - 5)^2), so both
    % ends land exactly on (4, -2) and (6, -2)
    \addplot[very thick, black, domain=0:pi, samples=100]
    ({5 + cos(x)}, {-2 + sin(x)});
    \addplot[very thick, black] coordinates {(6,2) (10,2)};
    % \node[below] at (axis cs:-2,3) {\footnotesize \(f(x)\)};
  \end{axis}
\end{tikzpicture}
\end{document}
