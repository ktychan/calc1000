%! TeX program = lualatex
% The two accumulation-function exercises in lessons/ftc-part-one.tex, where
% the reader reads intervals of increase and decrease of F(x) = int_c^x f(t)dt
% off the graph of f.
%
%   page 1 -- for F(x) = int_1^x f(t) dt, f drawn over [-0.5, 9]
%   page 2 -- for F(x) = int_{-3}^x f(t) dt, f drawn over [-3.5, 4]
%
% Each graph is built from three smooth pieces with two jumps between them,
% and the two jumps do different things on purpose:
%
%   * one jump crosses the axis, so f changes sign there.  F is still
%     continuous, but it turns around: a corner, and a local extremum at a
%     point where f is not even defined.
%   * one jump goes down without crossing, so f keeps its sign.  F carries on
%     in the same direction -- the slope just changes.  This is the one that
%     catches people out, and it is why both exercises have one.
%
% Every sign change of f -- smooth crossing or jump -- lands on an integer, so
% the answers are readable off the grid.  No open or filled dots are drawn at
% the jumps: a single point carries no net area, matching the convention in
% plot_definite_integral_area.
%
% Page 1.  f(t) = 2 sin(pi t/2)           on [-0.5, 3]   (0 at t = 0 and 2)
%          f(t) = 1.5 + 0.8 sin(pi(t-3))  on [3, 5]      (positive throughout)
%          f(t) = 0.8 cos(pi(t-5)/2)      on [5, 9]      (0 at t = 6 and 8)
% so f jumps from -2 up to 1.5 at t = 3 (across the axis) and from 1.5 down to
% 0.8 at t = 5 (not across).  F increases on (0,2), (3,6), (8,9) and decreases
% on (-0.5,0), (2,3), (6,8): local maxima at t = 2 and 6, local minima at
% t = 0, 3 and 8, and nothing at t = 5.
%
% Page 2.  f(t) = 1.5 cos(pi(t+1)/2)      on [-3.5, -1]  (0 at t = -2)
%          f(t) = -1.3 + 0.6 sin(pi(t+1)) on [-1, 2]     (negative throughout)
%          f(t) = -2.2 cos(pi(t-2)/2)     on [2, 4]      (0 at t = 3)
% so f jumps from 1.5 down to -1.3 at t = -1 (across the axis) and from -1.3
% down to -2.2 at t = 2 (not across).  F decreases on (-3.5,-2) and (-1,3) and
% increases on (-2,-1) and (3,4): a local minimum at t = -2 and t = 3, a local
% maximum at t = -1, and nothing at t = 2.  The window starts at -3.5 rather
% than -4 so the curve does not begin on a zero sitting under the "-4" tick
% label, and so that the lower limit t = -3 is inside the picture.
%
% 1.25cm cells: the sign changes are steep, and at smaller cells the curve
% clips the tick label it crosses.  Include at natural size -- no width=, and
% no scale= in \includestandalone.
\documentclass[tikz, crop]{standalone}

\input{../typography.tex.preamble}
\input{../tikz.tex.preamble}

% Matches the other exercise grids in these notes (plot-riemann-negative).
\pgfplotsset{accumulation grid/.style={
    axis lines=middle,
    axis equal image=false,
    x=1.25cm, y=1.25cm,
    xlabel={\(t\)}, xlabel style={anchor=west},
    ylabel={\(f(t)\)}, ylabel style={anchor=south},
    label style={at={(ticklabel* cs:1), font=\footnotesize}},
    % Every sign change sits on a labelled tick, so a steep crossing would run
    % straight through the digit.  axis on top draws the ticks and labels
    % after the curve and each label is backed with white, which keeps the
    % number readable; the grid is pushed back onto the background layer so
    % that it alone stays behind the curve.
    set layers,
    axis on top,
    grid=major, grid style={gray!20, /pgfplots/on layer=axis background},
    tick label style={fill=white, inner sep=0.8pt},
    ytick={-2,-1,1,2},
    ymin=-2.7, ymax=2.7,
    smooth, samples=120, no markers,
    clip=false,
  }}

\begin{document}

% ---------------------------------------------------------------------
% Page 1 -- for the exercise with lower limit 1.
\begin{tikzpicture}
  \begin{axis}[
    accumulation grid,
    xmin=-1.1, xmax=9.9,
    xtick={1,...,9},
    ]
    \addplot[very thick, domain=-0.5:3] {2*sin(pi*x/2)};
    \addplot[very thick, domain=3:5]    {1.5 + 0.8*sin(pi*(x-3))};
    \addplot[very thick, domain=5:9]    {0.8*cos(pi*(x-5)/2)};
  \end{axis}
\end{tikzpicture}

% ---------------------------------------------------------------------
% Page 2 -- for the exercise with lower limit -3.
\begin{tikzpicture}
  \begin{axis}[
    accumulation grid,
    xmin=-4.1, xmax=4.9,
    xtick={-3,-2,-1,1,2,3,4},
    ]
    \addplot[very thick, domain=-3.5:-1] {1.5*cos(pi*(x+1)/2)};
    \addplot[very thick, domain=-1:2]    {-1.3 + 0.6*sin(pi*(x+1))};
    \addplot[very thick, domain=2:4]     {-2.2*cos(pi*(x-2)/2)};
  \end{axis}
\end{tikzpicture}

\end{document}
