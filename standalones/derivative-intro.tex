% A balloon dropped from rest: height 4 - t^2, the same model as the s(t)
% picture further down, so the intro and the velocity discussion share a story.
\begin{tikzpicture}[scale=1]
  \begin{axis}[basic curve, xlabel={time \(t\)}, ylabel={\(f(t)\)},
               xticklabels={\empty}, yticklabels={\empty}, clip=false,
               xmin=0, xmax=2, ymin=0, ymax=4, enlargelimits]
    \addplot[very thick, domain=0:2] {-x^2 + 4};
    \fill[main] (axis cs:1,3) circle[radius=3pt];
    \node[main, below] at (axis cs:1,0) {\(a\)};
    \node[main, left] at (axis cs:0,3) {\(f(a)\)};
    \addplot[main, dashed] coordinates {(1,0) (1,3) (0,3)};
  \end{axis}
\end{tikzpicture}

% The same drop, but something catches the balloon at time a and it descends
% steadily afterwards: the speed changes abruptly, so the graph has a corner.
\begin{tikzpicture}[scale=1]
  \begin{axis}[basic curve, xlabel={time \(t\)}, ylabel={\(g(t)\)},
               xticklabels={\empty}, yticklabels={\empty}, clip=false,
               xmin=0, xmax=2, ymin=0, ymax=4, enlargelimits]
    \addplot[very thick, domain=0:1] {-x^2 + 4};
    \addplot[very thick, domain=1:2] {3 - (x - 1)/2};
    \fill[main] (axis cs:1,3) circle[radius=3pt];
    \node[main, below] at (axis cs:1,0) {\(a\)};
    \node[main, left] at (axis cs:0,3) {\(g(a)\)};
    \addplot[main, dashed] coordinates {(1,0) (1,3) (0,3)};
  \end{axis}
\end{tikzpicture}
