% One function carries both derivative tests, so the six pages below are six
% views of the same curve.
%
%   f(x)   = 0.215 (x^5/5 - x^4/4 - 2x^3) + 0.5
%   f'(x)  = 0.215 x^2 (x+2)(x-3)
%   f''(x) = 0.215 x (4x^2 - 3x - 12)
%
% f' has a DOUBLE root at 0.  That is the whole reason for a quintic: f'
% changes sign at -2 and at 3 but only touches zero at 0, so the three
% critical numbers A, B, C give one of each case of the first derivative test
% -- a local maximum, a point that is NEITHER, and a local minimum.  A quartic
% cannot do it: its derivative is a cubic, and a cubic with a double root has
% only two distinct roots, so one of the three would be lost.
%
% The same double root makes 0 an inflection point as well, which is why B
% appears on both sets of pages: it is the point where the tangent is
% horizontal and the curve still passes straight through.  The other two
% inflection points are the roots of 4x^2 - 3x - 12, namely (3 -/+ sqrt(201))/8
% = -1.3972 and 2.1472, written out below because pgfplots cannot be asked for
% them.
%
% The scale 0.215 and the lift 0.5 are chosen together to place A near the top
% of the window and C clear of the bottom; a further lift of 4.7 puts them at
% A = 6.404, B = 5.200, C = -0.315, against a window from -4.8 to 7, which
% also keeps the inflection point E = 1.763 clear of the x-axis.  The two
% concavity captions that sit BELOW the axis do not travel with the curve:
% there is no room under the axis to move them into, so they stay put while
% the two above it rise with the graph.
\pgfplotsset{
  derivative test/.style = {
    ymin=-4.8, ymax=7,
    % the axis runs wider than the plotted domain (-3.3 to 3.9), which
    % squishes the drawing and leaves margins for the labels at either end
    xmin=-3.6, xmax=4.4,
    xticklabel={\empty},
    yticklabel={\empty},
    width={200pt},
  }
}

\newcommand{\curvefp}{0.215*(x^4 - x^3 - 6*x^2)}

\newcommand{\dottedat}[1]{\addplot[dotted, thin] coordinates { (#1, -10) (#1, 10) };}
\newcommand{\markcrit}{\dottedat{-2}\dottedat{0}\dottedat{3}}
\newcommand{\markinfl}{\dottedat{-1.3972}\dottedat{0}\dottedat{2.1472}}

% f cut at its critical numbers and coloured by the sign of f'
\newcommand{\fbyslope}{%
  \addplot[ultra thick, smooth, samples=200, domain=-3.3:-2, main] {\curvef};
  \addplot[ultra thick, smooth, samples=200, domain=-2:3, supp] {\curvef};
  \addplot[ultra thick, smooth, samples=200, domain=3:3.9, main] {\curvef};}

% the same curve cut at its inflection points and coloured by the sign of f''
\newcommand{\fbyconcavity}{%
  \addplot[ultra thick, smooth, samples=200, domain=-3.3:-1.3972, supp] {\curvef};
  \addplot[ultra thick, smooth, samples=200, domain=-1.3972:0, main] {\curvef};
  \addplot[ultra thick, smooth, samples=200, domain=0:2.1472, supp] {\curvef};
  \addplot[ultra thick, smooth, samples=200, domain=2.1472:3.9, main] {\curvef};}
